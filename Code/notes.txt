#include<bits/stdc++.h>
using namespace std;

#define nl '\n'
#define i128 __int128
#define int long long
#define sz(x) ((int)(x).size())
#define no cout << "NO" << '\n'
#define yes cout << "YES" << '\n'
#define all(x) (x).begin(), (x).end()
#define allr(x) (x).rbegin(), (x).rend()
#define CEIL(a, b) (((a) + (b) - 1) / (b))
#define trace(x) cout << #x << ": " << x << " \n"
#define print(x) for (auto u : (x)) cout << u << ' '; cout << nl;
#define vecin(name, len) vector<int> name(len); for (auto &_ : name) cin >> _;



void solve() 
{
  int n; cin >> n;
  vector<vector<int>> v(n);
  int sz = 0;
  for (int i = 0; i < n; i++) {
    int l; cin >> l;
    sz += l;
    vector<int> temp;
    for (int j = 0; j < l; j++) {
      int x; cin >> x;
      temp.push_back(x);
    }
    int sum = accumulate(all(temp), 0LL);
    v[i].push_back(sum);
    for (auto u : temp) {
      v[i].push_back(u);
    }
  }

  sort(allr(v));
  trace(sz); 
  map<int, int> mp;

  int x = 1;

  for (int i = 0; i < n; i++) {
    for (auto u : v[i]) {
      cout << u << ' ';
      mp[u] = x;
      x++;
    }
    cout << nl;
  }
  vector<pair<int, int>> vp;
  for (auto [v, i] : mp) {
    vp.push_back({i, v});
  }
  sort(allr(vp));
  for (auto [i, v] : vp) {
    cout << i << ' ' << v << nl;
  }
}

int32_t main()
{
  ios_base::sync_with_stdio(false);
  cin.tie(NULL);

  int t = 1;
  cin >> t;
  while(t--) solve();

  return 0;
}
